import streamlit as st
import pandas as pd
import folium
from streamlit_folium import st_folium
import math

def haversine(lat1, lon1, lat2, lon2):
    R = 6371000
    phi1, phi2 = math.radians(lat1), math.radians(lat2)
    dphi = math.radians(lat2 - lat1)
    dlambda = math.radians(lon2 - lon1)
    a = math.sin(dphi/2)**2 + math.cos(phi1)*math.cos(phi2)*math.sin(dlambda/2)**2
    return 2 * R * math.asin(math.sqrt(a))

st.set_page_config(page_title="대구 문화유산 대중교통 안내", layout="wide")
st.title("🏛️ 대구 문화유산 ↔ 대중교통 연계 안내")

heritage_df = pd.read_csv("heritage.csv")
subway_df = pd.read_csv("subway.csv")
bus_df = pd.read_csv("busstop.csv")

selected = st.sidebar.selectbox("문화유산 선택", heritage_df["name"].tolist())
top_n = st.sidebar.slider("버스정류장 대안 노선 개수", 1, 9, 3)

row = heritage_df[heritage_df["name"] == selected].iloc[0]
h_lat, h_lon = float(row["lat"]), float(row["lon"])

st.subheader(selected)
st.write(f"**개방시간**: {row.get('hours','확인 필요')}")
st.write(f"**소개**: {row.get('desc','-')}")

subway_df["dist"] = subway_df.apply(lambda r: haversine(h_lat, h_lon, r["lat"], r["lon"]), axis=1)
bus_df["dist"] = bus_df.apply(lambda r: haversine(h_lat, h_lon, r["lat"], r["lon"]), axis=1)

nearest_subway = subway_df.nsmallest(2, "dist")
nearest_bus = bus_df.nsmallest(top_n, "dist")

col1, col2 = st.columns([2, 1])

with col1:
    m = folium.Map(location=[h_lat, h_lon], zoom_start=14)
    folium.Marker([h_lat, h_lon], popup=selected, icon=folium.Icon(color="red", icon="star")).add_to(m)
    for _, s in nearest_subway.iterrows():
        folium.Marker([s["lat"], s["lon"]], popup=f"{s['station']}역",
                      icon=folium.Icon(color="blue", icon="train")).add_to(m)
    for _, b in nearest_bus.iterrows():
        folium.Marker([b["lat"], b["lon"]], popup=f"{b['name']} / {b.get('routes','')}",
                      icon=folium.Icon(color="green", icon="bus", prefix="fa")).add_to(m)
    st_folium(m, width=800, height=600)

with col2:
    st.markdown("### 🚇 가까운 지하철역")
    for _, s in nearest_subway.iterrows():
        st.write(f"- {s['station']}역 (약 {s['dist']:.0f}m)")
    st.markdown("### 🚌 가까운 버스정류장")
    for _, b in nearest_bus.iterrows():
        st.write(f"- {b['name']} (약 {b['dist']:.0f}m) — {b.get('routes','정보없음')}")
